function renderCheckboxGroup(containerId, opzioni, selezionate) {
  const cont = document.getElementById(containerId);
  cont.innerHTML = '';
  opzioni.forEach((op, i) => {
    const id = containerId + '-' + i;
    const checked = selezionate.includes(op) ? 'checked' : '';
    cont.insertAdjacentHTML('beforeend',
      '<label style="display:flex;align-items:center;gap:6px;background:rgba(255,255,255,0.05);padding:6px 10px;border-radius:8px;cursor:pointer;font-size:13px;color:#fff;">' +
      '<input type="checkbox" value="' + op + '" id="' + id + '" ' + checked + '> ' + op +
      '</label>'
    );
  });
}

function apriProfiloTrasportatore() {
  document.getElementById('sezione-azienda').style.display = 'none';
  document.getElementById('sezione-profilo-trasportatore').style.display = 'block';
  const camionSel = trasportatoreFieldsCorrenti.tipo_camion || [];
  const autSel = trasportatoreFieldsCorrenti.autorizzazioni || [];
  renderCheckboxGroup('checkbox-tipo-camion', TIPO_CAMION_OPZIONI, camionSel);
  renderCheckboxGroup('checkbox-autorizzazioni', AUTORIZZAZIONI_OPZIONI, autSel);
  document.getElementById('sezione-profilo-trasportatore').scrollIntoView({behavior:'smooth', block:'start'});
}

async function salvaProfiloTrasportatore() {
  const mezziSelezionati = Array.from(document.querySelectorAll('#checkbox-tipo-camion input:checked')).map(el => el.value);
  const autSelezionate = Array.from(document.querySelectorAll('#checkbox-autorizzazioni input:checked')).map(el => el.value);
  try {
    await fetch(PROXY_WEBHOOK, {
      method: 'POST',
      headers: { 'Content-Type': 'application/json' },
      body: JSON.stringify({
        action: 'salva_profilo_trasportatore',
        record_id: trasportatoreRecordId,
        tipo_camion: mezziSelezionati,
        autorizzazioni: autSelezionate
      })
    });
    trasportatoreFieldsCorrenti.tipo_camion = mezziSelezionati;
    trasportatoreFieldsCorrenti.autorizzazioni = autSelezionate;
    document.getElementById('profilo-salvato-msg').style.display = 'inline';
    setTimeout(() => { document.getElementById('profilo-salvato-msg').style.display = 'none'; }, 3000);
  } catch(e) {
    console.error('Errore salvataggio profilo:', e);
    alert('Errore nel salvataggio, riprova tra un momento.');
  }
}
